data ABNV a = Hoja a | Uni a (ABNV a) | Bi (ABNV a) a (ABNV a)

foldABNV :: (a -> b) -> (a -> b -> b) -> (b -> a -> b -> b) -> ABNV a -> b
foldABNV fHoja fUni fBi t = case t of
  Hoja k -> fHoja k
  Uni k t' -> fUni k (rec t')
  Bi i k d -> fBi (rec i) k (rec d)
  where rec = foldABNV fHoja fUni fBi

recABNV :: (a -> b) -> (a -> ABNV a -> b -> b) -> (ABNV a -> b -> a -> ABNV a -> b -> b) -> ABNV a -> b
recABNV fHoja fUni fBi t = case t of
  Hoja k -> fHoja k
  Uni k t' -> fUni k t' (rec t')
  Bi i k d -> fBi i (rec i) k d (rec d)
  where rec = recABNV fHoja fUni fBi

elemABNV :: Eq a => a -> ABNV a -> Bool
elemABNV x = foldABNV (== x) (\k rec -> k == x || rec) (\i k d -> k == x || i || d)
